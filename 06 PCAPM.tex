\begin{frame}{The Production-Based Framework: Four Ingredients}
	\begin{enumerate}
		\item \textbf{Technology:} Cobb--Douglas $X_t=Z_tK_t^\alpha L_t^{1-\alpha}$ (constant returns to scale).
		\item \textbf{Adjustment costs:} convex $\Phi_t=\phi(I_t/K_t)K_t$, with $\phi'>0,\ \phi''\ge 0$.
		\item \textbf{Optimization:} the firm maximizes $V_t=\max \mathrm{E}_t\sum_{s\ge0}M_{t,t+s}D_{t+s}$.
		\item \textbf{Result:} investment $I/K$ and profitability $Y/K$ pin down expected returns.
	\end{enumerate}
	\vspace{4pt}
	Exposition follows \textcolor{blue}{\href{https://doi.org/10.1146/annurev-financial-110311-101731}{Kogan and Papanikolaou (2012)}}; structural estimation follows \textcolor{blue}{\href{https://doi.org/10.1086/649760}{Liu, Whited, and Zhang (2009)}}.
	\begin{block}{How to read the math}
		First we \textbf{build the machine} (technology $\to$ FOC $\to$ investment return), then \textbf{turn it into testable returns}, and finally deliver \textbf{Zhang's cross-sectional findings} and the $q$-factor model.
	\end{block}
\end{frame}

\section{Building Blocks: Technology and Optimization}

\begin{frame}{Technology and the Optimal Labor Choice}
	\textbf{Cobb--Douglas production (constant returns to scale):}
	\begin{equation}\label{equ:production}
		X_t = Z_t K_t^\alpha L_t^{1-\alpha},\qquad \alpha\in(0,1).
	\end{equation}
	\textbf{Labor is chosen statically} (take $K_t,\omega_t$ as given), so its FOC sets the marginal product of labor equal to the wage:
	\begin{equation*}
		\max_{L_t}\{X_t-\omega_t L_t\}\ \Rightarrow\ \frac{\partial X_t}{\partial L_t}=(1-\alpha)Z_tK_t^\alpha L_t^{-\alpha}=\omega_t.
	\end{equation*}
	Solving for labor demand:
	\begin{equation}\label{equ:optimal labor supply}
		L_t^*=\left(\frac{(1-\alpha)Z_t}{\omega_t}\right)^{1/\alpha}K_t.
	\end{equation}
	\vspace{2pt}
	\textit{Why we bother.} Substituting $L_t^*$ back turns the two-input technology into a \textbf{one-state} (capital-only) problem and makes profit proportional to the marginal product of capital --- the object that prices returns.
\end{frame}

\begin{frame}{Profitability Equals the Capital Share of Output}
	\textbf{Output at optimal labor} --- substitute (\ref{equ:optimal labor supply}) into (\ref{equ:production}):
	\begin{equation*}
		\begin{aligned}
			Y_t &= Z_t K_t^\alpha (L_t^*)^{1-\alpha}
			= Z_t K_t^\alpha\left[\left(\tfrac{(1-\alpha)Z_t}{\omega_t}\right)^{1/\alpha}K_t\right]^{1-\alpha}\\
			&= Z_t^{1/\alpha}\left(\frac{1-\alpha}{\omega_t}\right)^{(1-\alpha)/\alpha}K_t
			\;\Rightarrow\; \frac{Y_t}{K_t}=Z_t^{1/\alpha}\left(\frac{1-\alpha}{\omega_t}\right)^{(1-\alpha)/\alpha}.
		\end{aligned}
	\end{equation*}
	\textbf{Gross profit after paying labor:}
	\begin{equation}\label{equ: gross profit}
		\Pi_t=Y_t-\omega_t L_t^* = Y_t-(1-\alpha)\underbrace{Z_t^{1/\alpha}\!\left(\tfrac{1-\alpha}{\omega_t}\right)^{(1-\alpha)/\alpha}K_t}_{=\,Y_t}=\alpha Y_t.
	\end{equation}
	\begin{block}{Two facts we carry forward}
		(i) Profit $\Pi_t=\alpha Y_t$ --- capital's share of output ($\alpha\approx 1/3$ in US data). \\
		(ii) Output is linear in $K_t$, so the \textbf{marginal product of capital equals average profitability}: $\partial Y_t/\partial K_t = Y_t/K_t$. This ``profitability'' sits in the numerator of every return formula below.
	\end{block}
\end{frame}

\section{Investment Decision with Adjustment Costs}

\begin{frame}{Capital Accumulation and Adjustment Costs}
	\textbf{Law of motion for capital:}
	\begin{equation}\label{equ:capital accumulation}
		K_{t+1}=I_t+(1-\delta)K_t,\qquad \delta\in(0,1).
	\end{equation}
	\textbf{Total investment cost} (constant returns to scale in $(I_t,K_t)$):
	\begin{equation}\label{equ:investment cost}
		\Phi_t=\phi\!\left(\frac{I_t}{K_t}\right)K_t,\qquad \phi'>0,\ \phi''\ge 0,\ \phi(0)=0.
	\end{equation}
	\textbf{Why adjustment costs are essential.}
	\begin{itemize}
		\item Without them (Hayashi: $\phi(I/K)=I/K$, so $\Phi_t=I_t$), capital jumps instantly to target and investment is infinitely elastic to $q$ --- counterfactual.
		\item Convex costs (installation, disruption, retraining, resale discounts) make investment \textbf{sluggish and forward-looking}, and give $\phi'(\cdot)$ a non-trivial slope --- the engine of return variation.
	\end{itemize}
\end{frame}

\begin{frame}{Dividends and Firm Value}
	\textbf{Dividend (cash flow to shareholders):}
	\begin{equation}\label{equ:dividend}
		D_t=\Pi_t-\Phi_t=\alpha Y_t-\phi\!\left(\frac{I_t}{K_t}\right)K_t.
	\end{equation}
	\textbf{Firm value (present value of dividends):}
	\begin{equation}\label{equ:firm value}
		V_t=\mathrm{E}_t\sum_{s=0}^{\infty}M_{t,t+s}D_{t+s},
	\end{equation}
	where $M_{t,t+s}$ is the stochastic discount factor from $t$ to $t+s$.
	\begin{itemize}
		\item The manager chooses $I_t$ to maximize $V_t$.
		\item Current investment affects \textit{all} future cash flows through capital accumulation (\ref{equ:capital accumulation}) --- the source of the dynamic trade-off.
	\end{itemize}
\end{frame}

\section{The Firm's Optimization Problem}

\begin{frame}{Dynamic Programming Formulation}
    \textbf{Bellman Equation:}
    \begin{equation}\label{equ:bellman}
        V_t = \max_{I_t} \left\{ D_t + \mathrm{E}_t\left[M_{t+1} V_{t+1}\right] \right\}
    \end{equation}
    
    \textbf{Interpretation:}
    \begin{itemize}
        \item Today's value = Today's dividend + Discounted expected future value
        \item Choice variable: Investment $I_t$
        \item Trade-off:
        \begin{itemize}
            \item Higher $I_t$ $\rightarrow$ Lower $D_t$ today (costs money)
            \item But higher $K_{t+1}$ $\rightarrow$ Higher $V_{t+1}$ tomorrow (more production)
        \end{itemize}
    \end{itemize}
    
    
    \textbf{First-Order Condition:}
    \begin{equation}\label{equ:foc}
        \frac{\partial D_t}{\partial I_t} + \mathrm{E}_t\left[M_{t+1} \frac{\partial V_{t+1}}{\partial I_t}\right] = 0
    \end{equation}
    
    Using $\frac{\partial D_t}{\partial I_t} = -\phi'\left(\frac{I_t}{K_t}\right)$ and $\frac{\partial V_{t+1}}{\partial I_t} = \frac{\partial V_{t+1}}{\partial K_{t+1}}$:
    \begin{equation}\label{equ:marginal}
        \phi'\left(\frac{I_t}{K_t}\right) = \mathrm{E}_t\left[M_{t+1} \frac{\partial V_{t+1}}{\partial K_{t+1}}\right]
    \end{equation}
\end{frame}

\begin{frame}[allowframebreaks]{Interpreting the First-Order Condition}
    \begin{equation*}
        \underbrace{\phi'\left(\frac{I_t}{K_t}\right)}_{\text{Marginal cost of investing}} = \underbrace{\mathrm{E}_t\left[M_{t+1} \frac{\partial V_{t+1}}{\partial K_{t+1}}\right]}_{\text{Expected discounted marginal benefit}}
    \end{equation*}
    
    
    \textbf{Left-Hand Side (Marginal Cost):}
    \begin{itemize}
        \item Cost of investing one more unit today
        \item Includes purchase price + adjustment costs
        \item Reduces current dividends
    \end{itemize}
    
    
    \textbf{Right-Hand Side (Marginal Benefit):}
    \begin{itemize}
        \item Value of having one more unit of capital tomorrow
        \item Discounted by SDF $M_{t+1}$ (adjust for risk and time preference)
        \item Increases future production and profits
    \end{itemize}
    
    
    \textbf{Equilibrium:}
    \begin{itemize}
        \item Invest until marginal cost = expected marginal benefit
        \item If $\phi' < \mathrm{E}[M V']$: Invest more (benefit exceeds cost)
        \item If $\phi' > \mathrm{E}[M V']$: Invest less (cost exceeds benefit)
    \end{itemize}
\end{frame}

\begin{frame}{The Envelope Theorem}
    \textbf{Goal:} Derive the marginal value of capital $\frac{\partial V_{t+1}}{\partial K_{t+1}}$.

    By the envelope theorem, differentiate the Bellman equation with respect to the state variable, holding the optimal investment choice fixed; then advance one period:
    \begin{equation}\label{equ:envelope}
        \frac{\partial V_{t+1}}{\partial K_{t+1}} = \underbrace{\frac{\partial D_{t+1}}{\partial K_{t+1}}}_{\text{Step 1: current dividend}} + \underbrace{\mathrm{E}_{t+1}\left[M_{t+2} \frac{\partial V_{t+2}}{\partial K_{t+1}}\right]}_{\text{Step 2--3: continuation value}}
    \end{equation}

    \textbf{Step 1: Differentiate the dividend} $D_{t+1} = \alpha Y_{t+1} - \Phi_{t+1}$, with $\Phi_{t+1} \equiv \phi\!\left(\frac{I_{t+1}}{K_{t+1}}\right) K_{t+1}$:
    \begin{equation*}
        \frac{\partial D_{t+1}}{\partial K_{t+1}} = \alpha \frac{\partial Y_{t+1}}{\partial K_{t+1}} - \frac{\partial \Phi_{t+1}}{\partial K_{t+1}}
    \end{equation*}
\end{frame}

\begin{frame}{The Envelope Theorem: Cost Term}
    \textbf{Step 1 continued:} Differentiate the investment-cost term.
    
    Apply the \textbf{product rule} to $\Phi_{t+1} = \phi(I_{t+1}/K_{t+1}) \cdot K_{t+1}$:
    \begin{equation*}
        \begin{aligned}
            \frac{\partial \Phi_{t+1}}{\partial K_{t+1}} &= \underbrace{\phi'\!\left(\tfrac{I_{t+1}}{K_{t+1}}\right) \cdot \left(-\tfrac{I_{t+1}}{K_{t+1}^2}\right) \cdot K_{t+1}}_{\text{chain rule on } \phi(I/K)} + \underbrace{\phi\!\left(\tfrac{I_{t+1}}{K_{t+1}}\right) \cdot 1}_{\text{derivative of } K} \\
            &= \phi\!\left(\tfrac{I_{t+1}}{K_{t+1}}\right) - \phi'\!\left(\tfrac{I_{t+1}}{K_{t+1}}\right) \tfrac{I_{t+1}}{K_{t+1}}
        \end{aligned}
    \end{equation*}

    \vspace{4pt}
    \textbf{Why this matters:} holding investment fixed, a larger capital base lowers the investment rate $I/K$ and therefore changes adjustment costs.
\end{frame}

\begin{frame}{The Envelope Theorem (cont.)}
    \textbf{Step 2: Use Chain Rule on Future Value Term}
    \begin{equation*}
        \frac{\partial V_{t+2}}{\partial K_{t+1}} = \frac{\partial V_{t+2}}{\partial K_{t+2}} \cdot \frac{\partial K_{t+2}}{\partial K_{t+1}} = \frac{\partial V_{t+2}}{\partial K_{t+2}} (1-\delta)
    \end{equation*}
    
    because $K_{t+2} = I_{t+1} + (1-\delta)K_{t+1}$ and the envelope step holds $I_{t+1}$ fixed (only undepreciated capital matters)
    
    
    \textbf{Step 3: Apply FOC (\ref{equ:marginal}) One Period Forward}
    \begin{equation*}
        \mathrm{E}_{t+1}\left[M_{t+2} \frac{\partial V_{t+2}}{\partial K_{t+2}}\right] = \phi'\left(\frac{I_{t+1}}{K_{t+1}}\right)
    \end{equation*}
    
    
    \textbf{Final Result (Envelope Theorem):}
    \begin{equation}\label{equ:envelope theorem}
        \boxed{
        \frac{\partial V_{t+1}}{\partial K_{t+1}} = \alpha \frac{\partial Y_{t+1}}{\partial K_{t+1}} - \frac{\partial \Phi_{t+1}}{\partial K_{t+1}} + (1-\delta) \phi'\left(\frac{I_{t+1}}{K_{t+1}}\right)
        }
    \end{equation}
\end{frame}

\begin{frame}{Economic Interpretation of the Envelope Theorem}
    \begin{equation*}
        \frac{\partial V_{t+1}}{\partial K_{t+1}} = \underbrace{\alpha \frac{\partial Y_{t+1}}{\partial K_{t+1}}}_{\text{(1) Profit contribution}} - \underbrace{\frac{\partial \Phi_{t+1}}{\partial K_{t+1}}}_{\text{(2) Cost effect}} + \underbrace{(1-\delta) \phi'\left(\frac{I_{t+1}}{K_{t+1}}\right)}_{\text{(3) Scrap value}}
    \end{equation*}
    
    
    \textbf{Three Components:}
    \begin{enumerate}
        \item \textbf{Marginal Profit:} Extra output $\times$ profit share $\alpha$
        \begin{itemize}
            \item More capital $\rightarrow$ more production $\rightarrow$ more profits
        \end{itemize}
        
        \item \textbf{Cost Reduction:} Lower average adjustment costs
        \begin{itemize}
            \item Bigger capital base $\rightarrow$ same investment is smaller fraction
            \item Recall $\Phi = \phi(I/K) \cdot K$, so $\partial \Phi/\partial K < 0$ if $I > 0$
        \end{itemize}
        
        \item \textbf{Continuation Value:} Undepreciated capital carries forward
        \begin{itemize}
            \item Fraction $(1-\delta)$ survives to next period
            \item Worth its marginal cost of installation $\phi'$
        \end{itemize}
    \end{enumerate}
    
    \vspace{0.2cm}
    \textit{The marginal value of capital equals the sum of its immediate profit contribution and its continuation value}
\end{frame}

\section{Constant Returns to Scale: Tobin's Q}

\begin{frame}{Hayashi's (1982) Key Result}
    \textbf{Assumption: Constant Returns to Scale (CRS)}
    \begin{itemize}
        \item Production: $X_t = Z_t K_t^\alpha L_t^{1-\alpha}$ exhibits CRS
        \item Adjustment costs: $\Phi_t = \phi(I_t/K_t) K_t$ exhibits CRS in $(I_t, K_t)$
    \end{itemize}
    
    
    \textbf{Hayashi's Theorem:}
    
    Under CRS, marginal $q$ equals average $q$:
    \begin{equation}\label{equ:hayashi}
        \boxed{
        \frac{\partial V_t}{\partial K_t} = \frac{V_t}{K_t}
        }
    \end{equation}
    
    
    \textbf{Intuition:}
    \begin{itemize}
        \item CRS $\rightarrow$ doubling all inputs doubles output and value
        \item Marginal and average products are equal
        \item Value of firm = Marginal value of capital $\times$ Quantity of capital
        \item Key simplification: Don't need to estimate marginal $q$, just use $V/K$
    \end{itemize}
\end{frame}

\begin{frame}{Deriving Tobin's Q Equation}
    \textbf{Start with FOC (\ref{equ:marginal}):}
    \begin{equation*}
        \phi'\left(\frac{I_t}{K_t}\right) = \mathrm{E}_t\left[M_{t+1} \frac{\partial V_{t+1}}{\partial K_{t+1}}\right]
    \end{equation*}
    
    \textbf{Apply Hayashi result (\ref{equ:hayashi}):}
    \begin{equation*}
        \phi'\left(\frac{I_t}{K_t}\right) = \mathrm{E}_t\left[M_{t+1} \frac{V_{t+1}}{K_{t+1}}\right]
    \end{equation*}
    
    \textbf{Use Bellman equation: $V_t = D_t + \mathrm{E}_t[M_{t+1} V_{t+1}]$}
    
    Since $K_{t+1}$ is known at time $t$:
    \begin{equation*}
        \phi'\left(\frac{I_t}{K_t}\right) = \frac{\mathrm{E}_t[M_{t+1} V_{t+1}]}{K_{t+1}} = \frac{V_t - D_t}{K_{t+1}}
    \end{equation*}
    
    \textbf{Final Form (Tobin's Q):}
    \begin{equation}\label{equ:tobins q}
        \boxed{
        \phi'\left(\frac{I_t}{K_t}\right) = \frac{V_t - D_t}{K_{t+1}} = \frac{D_t}{K_{t+1}} \cdot \frac{V_t - D_t}{D_t}
        }
    \end{equation}
\end{frame}

\begin{frame}{Unpacking Tobin's Q}
	\begin{equation*}
		\phi'\!\left(\frac{I_t}{K_t}\right)=\underbrace{\frac{D_t}{K_{t+1}}}_{\text{current profitability}}\times\underbrace{\frac{V_t-D_t}{D_t}}_{\text{price--dividend ratio}}
	\end{equation*}
	\textbf{Reading the identity.}
	\begin{itemize}
		\item \textbf{LHS} --- marginal cost of investing today; rising in $I_t/K_t$ by convexity.
		\item \textbf{First RHS term} $D_t/K_{t+1}$ --- how profitable installed capital is \textit{right now}.
		\item \textbf{Second RHS term} $(V_t-D_t)/D_t$ --- growth opportunities priced by the market.
	\end{itemize}
	\vspace{2pt}
	\textbf{Economic content (the seed of the cross-section).}
	\begin{itemize}
		\item High $I/K \Rightarrow$ RHS high $\Rightarrow$ \textit{either} high current profitability \textit{or} rich valuation (strong expected growth), or both.
		\item So \textbf{investment behavior reveals} information about profitability and discount rates: we can read expected returns off observable investment. The \textbf{next slide} makes this precise.
	\end{itemize}
\end{frame}

\begin{frame}{The Direct Link to Returns: Campbell--Shiller}
	\textbf{Take logs of the Tobin's Q identity} (\ref{equ:tobins q}) and expand the resulting log price--dividend ratio with the \textbf{Campbell--Shiller} identity (lower case $=$ logs; $\rho\approx0.96$ the linearization constant):
	\begin{equation}\label{equ:cs link}
		\boxed{
		\log\phi'\!\left(\frac{I_t}{K_t}\right)
		=\underbrace{(d_t-k_{t+1})}_{\text{current profitability}}
		+\underbrace{\mathrm{E}_t\!\sum_{j=0}^{\infty}\rho^{j}\,\Delta d_{t+1+j}}_{\text{expected cash-flow growth}}
		-\underbrace{\mathrm{E}_t\!\sum_{j=0}^{\infty}\rho^{j}\,r_{t+1+j}}_{\text{expected returns}}
		}
	\end{equation}
	\textbf{Now returns enter explicitly.} The first factor of (\ref{equ:tobins q}) is log profitability $d_t-k_{t+1}$; the second --- the log price--dividend ratio --- splits into expected cash-flow growth \textit{minus} expected returns (the discount rate).
	\begin{block}{The investment effect, read straight off returns}
		Hold profitability $d_t-k_{t+1}$ and expected cash-flow growth fixed $\Rightarrow$
		\textbf{high $I_t/K_t \Leftrightarrow$ low expected returns} $\mathrm{E}_t\sum_{j}\rho^{j}r_{t+1+j}$.
		Firms invest more exactly when their \textbf{discount rate is low} --- the microfoundation of the \textbf{investment factor} (low$-$minus$-$high $I/A$). \hfill{\footnotesize Cochrane, \textit{Asset Pricing}, eq.~(7.15).}
	\end{block}
\end{frame}

\section{The \texorpdfstring{$q$}{q}-Factor Model: Hou, Xue, and Zhang (2015)}

\begin{frame}{HXZ (2015): A Self-Contained Two-Period Derivation}
	\textbf{Why this section.} The spine promised $\mathrm{E}[1+R^S]=\text{profitability}/(1+a\,I/K)$. \href{https://doi.org/10.1093/rfs/hhu068}{\textcolor{blue}{Hou, Xue, and Zhang (2015)}} deliver it in the \textbf{simplest possible} two-period general-equilibrium model (after \href{https://doi.org/10.1016/j.jmoneco.2013.01.001}{\textcolor{blue}{Lin and Zhang 2013}}), then turn it into a tradable four-factor model.
	\vspace{2pt}
	\textbf{Two dates $t=0,1$; representative household; firms $i=1,\dots,N$.}
	\begin{itemize}
		\item \textbf{Household} maximizes $U(C_0)+\rho\,\mathrm{E}_0[U(C_1)]$, which yields the SDF $M_1\equiv\rho\,U'(C_1)/U'(C_0)$ and the consumption first principle
		\[
			\mathrm{E}_0\!\left[M_1\,r^S_{i1}\right]=1,\qquad r^S_{i1}\equiv\frac{P_{i1}+D_{i1}}{P_{i0}}.
		\]
		\item \textbf{Firm $i$} starts with assets $A_{i0}$ and known profitability $\Pi_{i0}$; operating cash flow is $\Pi_{it}A_{it}$. Assets depreciate \textbf{fully} ($\delta=100\%$), so $A_{i1}=I_{i0}$. Investing incurs a quadratic adjustment cost $\frac{a}{2}(I_{i0}/A_{i0})^{2}A_{i0}$.
	\end{itemize}
	\textbf{Date-0 free cash flow (the dividend):}
	\[
		D_{i0}=\Pi_{i0}A_{i0}-I_{i0}-\tfrac{a}{2}\!\left(\tfrac{I_{i0}}{A_{i0}}\right)^{2}\!A_{i0}.
	\]
	{\footnotesize Notation maps to the multi-period model: assets $A\leftrightarrow K$, profitability $\Pi\leftrightarrow Y/K$.}
\end{frame}

\begin{frame}{Step 1 --- The Investment Euler Equation}
	\textbf{Firm chooses $I_{i0}$ to maximize cum-dividend value} (with $A_{i1}=I_{i0}$; date-1 is terminal, so $I_{i1}=0$, $P_{i1}=0$, $D_{i1}=\Pi_{i1}A_{i1}$):
	\begin{equation}\label{equ:hxz value}
		P_{i0}+D_{i0}=\max_{I_{i0}}\;\Pi_{i0}A_{i0}-I_{i0}-\frac{a}{2}\!\left(\frac{I_{i0}}{A_{i0}}\right)^{2}\!A_{i0}+\mathrm{E}_0\!\left[M_1\,\Pi_{i1}A_{i1}\right].
	\end{equation}
	\textbf{Differentiate w.r.t.\ $I_{i0}$} (using $\partial A_{i1}/\partial I_{i0}=1$):
	\[
		\underbrace{-1}_{\text{purchase price}}\;\underbrace{-\,a\,\frac{I_{i0}}{A_{i0}}}_{\text{marginal adj.\ cost}}\;+\;\underbrace{\mathrm{E}_0\!\left[M_1\,\Pi_{i1}\right]}_{\text{discounted marginal benefit}}=0.
	\]
	\textbf{First principle for investment} --- marginal cost $=$ marginal $q$:
	\begin{equation}\label{equ:hxz foc}
		\boxed{\;1+a\,\frac{I_{i0}}{A_{i0}}=\mathrm{E}_0\!\left[M_1\,\Pi_{i1}\right]\;}
	\end{equation}
	One extra unit of assets costs the price (unity) $+$ marginal adjustment cost $a\,I/A$; its discounted marginal benefit is next period's profitability $\Pi_{i1}$.
\end{frame}

\begin{frame}{Step 2 --- Stock Return Equals Investment Return}
	\textbf{Ex-dividend value at the optimum.} Subtract $D_{i0}$ from (\ref{equ:hxz value}):
	\[
		P_{i0}=\mathrm{E}_0\!\left[M_1\,\Pi_{i1}A_{i1}\right].
	\]
	\textbf{Form the realized stock return} ($P_{i1}=0$, $D_{i1}=\Pi_{i1}A_{i1}$; the date-0--known $A_{i1}$ cancels):
	\[
		r^S_{i1}=\frac{P_{i1}+D_{i1}}{P_{i0}}=\frac{\Pi_{i1}A_{i1}}{\mathrm{E}_0[M_1\Pi_{i1}A_{i1}]}=\frac{\Pi_{i1}}{\mathrm{E}_0[M_1\Pi_{i1}]}.
	\]
	\textbf{Substitute the FOC (\ref{equ:hxz foc})} into the denominator:
	\begin{equation}\label{equ:hxz invreturn}
		\boxed{\;r^S_{i1}=\frac{\Pi_{i1}}{1+a\,(I_{i0}/A_{i0})}\;\equiv\;r^I_{i1}\;}
	\end{equation}
	The \textbf{financial} return equals the \textbf{investment} return: next-period marginal benefit $\Pi_{i1}$ over this-period marginal cost $1+a\,I/A$. \;{\footnotesize(With debt and taxes, \href{https://doi.org/10.1086/649760}{\textcolor{blue}{LWZ 2009}} show $r^I=$ WACC.)}
\end{frame}

\begin{frame}{Step 3 --- The Investment CAPM and Its Two Channels}
	\textbf{Take date-0 expectations of (\ref{equ:hxz invreturn}):}
	\begin{equation}\label{equ:hxz icapm}
		\boxed{\;\mathrm{E}_0\!\left[r^S_{i1}\right]=\frac{\mathrm{E}_0[\Pi_{i1}]}{1+a\,(I_{i0}/A_{i0})}\;}
	\end{equation}
	A cross-sectional restriction linking \textbf{two observable characteristics} to expected returns.
	\begin{block}{Two channels (each holds the other characteristic fixed)}
		\begin{itemize}
			\item \textbf{Investment channel:} given $\mathrm{E}_0[\Pi_{i1}]$, expected return \textit{decreases} in $I/A$.\\
			{\footnotesize Low discount rate $\Rightarrow$ high marginal $q$ $\Rightarrow$ high investment. $\;\Rightarrow$ the \textbf{$I/A$ factor}.}
			\item \textbf{Profitability channel:} given $I/A$, expected return \textit{increases} in $\mathrm{E}_0[\Pi_{i1}]$.\\
			{\footnotesize High $\Pi$ alongside low $I/A$ can only be reconciled by a high discount rate. $\;\Rightarrow$ the \textbf{ROE factor}.}
		\end{itemize}
	\end{block}
	These two channels organize a huge set of anomalies: value, issuance, accruals, asset growth $\to$ \textit{investment}; momentum, PEAD, distress $\to$ \textit{profitability}.
\end{frame}

\begin{frame}{From Theory to Tradable Factors: Construction}
	\textbf{Sample} Jan 1972--Dec 2012; financials and negative-book-equity firms excluded.
	\begin{itemize}
		\item \textbf{Investment} $I/A=$ annual change in total assets (AT) $/$ lagged total assets.
		\item \textbf{Profitability} ROE $=$ income before extraordinary items (IBQ) $/$ 1-quarter-lagged book equity.
	\end{itemize}
	\textbf{Triple $2\times3\times3$ sort} on size, $I/A$, ROE (NYSE breakpoints, value-weighted) $\Rightarrow$ \textbf{18 portfolios}:
	\begin{itemize}
		\item $r_{ME}=$ 9 small $-$ 9 big;\quad $r_{I/A}=$ 6 low $-$ 6 high;\quad $r_{ROE}=$ 6 high $-$ 6 low.
		\item Size \& $I/A$ rebalanced annually; ROE rebalanced \textbf{monthly} on fresh quarterly earnings.
	\end{itemize}
	\textbf{The four-factor $q$-model:}
	\begin{equation}\label{equ:qmodel}
		\mathrm{E}[r_i]-r_f=\beta^i_{\text{MKT}}\mathrm{E}[\text{MKT}]+\beta^i_{ME}\mathrm{E}[r_{ME}]+\beta^i_{I/A}\mathrm{E}[r_{I/A}]+\beta^i_{ROE}\mathrm{E}[r_{ROE}].
	\end{equation}
\end{frame}

\section{The \texorpdfstring{$q^5$}{q5} Model: Augmenting with Expected Growth (HMXZ 2021)}

\begin{frame}{Why a Fifth Factor? From Static to Dynamic}
	\textbf{What HXZ (2015) priced.} Their \textbf{static} collapse (gross return $1+R^S\equiv r^S$, with $\delta=100\%$ over two periods) used \textbf{two} characteristics --- investment and expected profitability:
	\[
		\mathrm{E}_t[1+R^S_{i,t+1}]=\frac{\mathrm{E}_t[\Pi_{i,t+1}]}{1+a\,(I_{it}/A_{it})}.
	\]
	But across the broad cross-section, \textbf{52 of the 150 significant anomalies} in HXZ (2019) still escape the $q$-factor model: R\&D-to-market, operating \& discretionary accruals, earnings-announcement drift, $\dots$
	\begin{block}{The repair: restore the dynamics HXZ assumed away}
		Put depreciation $\delta<1$ and an \textbf{infinite horizon} back in, so that \textit{next} period's investment becomes a live margin. A \textbf{third determinant} of expected returns appears --- \textbf{expected investment growth}.
	\end{block}
	\href{https://doi.org/10.1093/rof/rfaa004}{\textcolor{blue}{Hou, Mo, Xue, and Zhang (2021, \textit{Review of Finance})}} --- the ``$q^5$'' model: the four $q$-factors $+$ an \textbf{expected growth factor} $R_{\text{Eg}}$.
\end{frame}

\begin{frame}{The Dynamic Firm Problem}
	\textbf{Infinite horizon}; a representative household delivers the SDF $M_{t+1}$ with the consumption first principle $\mathrm{E}_t[M_{t+1}(1+R^S_{i,t+1})]=1$. Firm $i$ (same notation as the HXZ section: $A\leftrightarrow$ assets, $\Pi\leftrightarrow$ profitability $=$ return on assets, operating profit $=\Pi_{it}A_{it}$):
	\begin{itemize}
		\item \textbf{Capital accumulation:} $A_{i,t+1}=I_{it}+(1-\delta)A_{it}$, \quad $\delta\in(0,1)$.
		\item \textbf{Quadratic adjustment cost:} $\dfrac{a}{2}\!\left(\dfrac{I_{it}}{A_{it}}\right)^{2}\!A_{it}$.
	\end{itemize}
	\textbf{Net payout (the dividend):}
	\[
		D_{it}=\Pi_{it}A_{it}-I_{it}-\frac{a}{2}\!\left(\frac{I_{it}}{A_{it}}\right)^{2}\!A_{it}.
	\]
	\textbf{The firm chooses $\{I_{is}\}_{s\ge t}$ to maximize cum-dividend equity value:}
	\[
		P_{it}+D_{it}=\max_{\{I_{is}\}}\ \mathrm{E}_t\!\sum_{s=0}^{\infty} M_{t,t+s}\,D_{i,t+s}.
	\]
	{\footnotesize Difference from HXZ's two-period model: $\delta<1$ keeps installed capital alive, and the infinite horizon makes \textbf{next period's} investment $I_{i,t+1}$ a margin --- the seed of the new factor.}
\end{frame}

\begin{frame}{Step 1 --- The Bellman Problem and the Chain-Rule FOC}
	\textbf{Recursive (Bellman) form.} The infinite-horizon problem on the previous slide is, written as the \textbf{ex-dividend} value $P_{it}$ --- pick $I_{it}$ today, behave optimally thereafter:
	\begin{equation}\label{equ:q5 bellman}
		P_{it}=\max_{I_{it}}\ \mathrm{E}_t\!\left[M_{t+1}\big(P_{i,t+1}+D_{i,t+1}\big)\right],
	\end{equation}
	where $M_{t+1}$ is the SDF and $P_{i,t+1}+D_{i,t+1}$ is next period's \textbf{cum-dividend} value.
	\vspace{2pt}
	\textbf{Choose $I_{it}$.} It enters \emph{today's} dividend $D_{it}$ directly, and \emph{tomorrow's} assets through $A_{i,t+1}=I_{it}+(1-\delta)A_{it}$ --- so investment converts one-for-one into capital, $\partial A_{i,t+1}/\partial I_{it}=1$. Differentiating (\ref{equ:q5 bellman}) with the chain rule and setting it to zero:
	\[
		\frac{\partial D_{it}}{\partial I_{it}}+\mathrm{E}_t\!\left[M_{t+1}\,\frac{\partial (P_{i,t+1}+D_{i,t+1})}{\partial A_{i,t+1}}\,\frac{\partial A_{i,t+1}}{\partial I_{it}}\right]=0.
	\]
	\textbf{The trade-off:} the first term is the marginal cost paid through today's payout; the second is the discounted marginal value of the extra installed capital.
\end{frame}

\begin{frame}{Step 1 --- From the FOC to the First Principle of Investment}
	\textbf{Marginal effect on the dividend.} From $D_{it}=\Pi_{it}A_{it}-I_{it}-\frac{a}{2}\,(I_{it}/A_{it})^{2}A_{it}$ (with $A_{it}$ predetermined),
	\[
		\frac{\partial D_{it}}{\partial I_{it}}=-\underbrace{1}_{\text{price}}-\underbrace{a\,\frac{I_{it}}{A_{it}}}_{\text{marg.\ adj.\ cost}}.
	\]
	\textbf{Substitute} this and $\partial A_{i,t+1}/\partial I_{it}=1$ into the chain-rule FOC. Raising $I_{it}$ by one unit costs the dividend above and buys one extra unit of $A_{i,t+1}$, worth its \textbf{marginal value of capital}:
	\[
		-\Big(1+a\,\frac{I_{it}}{A_{it}}\Big)+\mathrm{E}_t\!\left[M_{t+1}\,\partial (P_{i,t+1}{+}D_{i,t+1})/\partial A_{i,t+1}\right]=0.
	\]
	\textbf{First principle of investment} --- move the cost left: marginal cost $=$ discounted marginal value of capital:
	\begin{equation}\label{equ:q5 foc}
		\boxed{\;1+a\,\frac{I_{it}}{A_{it}}=\mathrm{E}_t\!\left[M_{t+1}\,\partial (P_{i,t+1}{+}D_{i,t+1})/\partial A_{i,t+1}\right]\;}
	\end{equation}
	The next two steps compute this marginal value of capital explicitly.
\end{frame}

\begin{frame}{Step 2 --- The Marginal Value of Capital (Envelope)}
	\textbf{Differentiate next period's cum-dividend value w.r.t.\ $A_{i,t+1}$}, holding the optimal $I_{i,t+1}$ fixed (envelope):
	\[
		\frac{\partial(P_{i,t+1}+D_{i,t+1})}{\partial A_{i,t+1}}=\underbrace{\frac{\partial D_{i,t+1}}{\partial A_{i,t+1}}}_{\text{(i) current}}+\underbrace{\mathrm{E}_{t+1}\!\left[M_{t+2}\,\frac{\partial(P_{i,t+2}+D_{i,t+2})}{\partial A_{i,t+2}}\,\frac{\partial A_{i,t+2}}{\partial A_{i,t+1}}\right]}_{\text{(ii) continuation}}.
	\]
	\textbf{(i) Dividend term} --- differentiate $D_{i,t+1}=\Pi_{i,t+1}A_{i,t+1}-I_{i,t+1}-\frac{a}{2}\,I_{i,t+1}^2/A_{i,t+1}$:
	\[
		\frac{\partial D_{i,t+1}}{\partial A_{i,t+1}}=\underbrace{\Pi_{i,t+1}}_{\text{marg.\ profit}}+\underbrace{\frac{a}{2}\!\left(\frac{I_{i,t+1}}{A_{i,t+1}}\right)^{2}}_{\text{adj.-cost saving}},
		\qquad\text{since}\ \frac{\partial}{\partial A}\!\Big(\!-\tfrac{a}{2}\tfrac{I^2}{A}\Big)=+\tfrac{a}{2}\tfrac{I^2}{A^2}.
	\]
	\textbf{(ii) Continuation term} --- $\partial A_{i,t+2}/\partial A_{i,t+1}=1-\delta$, and the FOC (\ref{equ:q5 foc}) \textbf{one period forward} gives $\mathrm{E}_{t+1}[M_{t+2}\,\partial(\cdots)/\partial A_{i,t+2}]=1+a\,I_{i,t+1}/A_{i,t+1}$, so this term is
	\[
		(1-\delta)\Big(1+a\,\frac{I_{i,t+1}}{A_{i,t+1}}\Big).
	\]
\end{frame}

\begin{frame}{Step 3 --- The Multiperiod Investment Return}
	\textbf{Collect the marginal value of capital} (Step 2) and divide by the marginal cost (Step 1). With $1+R^S=1+R^I$ period-by-period (Cochrane 1991; Restoy--Rockinger 1994, under CRS):
	\begin{equation}\label{equ:q5 ir}
		\boxed{\;
		1+R^S_{i,t+1}=\frac{\;\Pi_{i,t+1}+\dfrac{a}{2}\!\left(\dfrac{I_{i,t+1}}{A_{i,t+1}}\right)^{2}+(1-\delta)\!\left(1+a\,\dfrac{I_{i,t+1}}{A_{i,t+1}}\right)\;}{1+a\,(I_{it}/A_{it})}\;}
	\end{equation}
	\begin{block}{Nesting the static model}
		Set \textbf{next-period} investment to zero ($I_{i,t+1}=0$): the numerator collapses to $\Pi_{i,t+1}+1-\delta$ --- the general static return. HXZ additionally took $\delta=100\%$ to reach the bare $\Pi_{i,t+1}$. \textbf{All the new content lives in the surviving $I_{i,t+1}/A_{i,t+1}$ terms.}
	\end{block}
	{\footnotesize With debt and taxes the denominator becomes the weighted average cost of capital (\href{https://doi.org/10.1086/649760}{\textcolor{blue}{LWZ 2009}}); (\ref{equ:q5 ir}) is exactly the NPV rule of capital budgeting.}
\end{frame}

\begin{frame}{Three Determinants: Reading the Cross-Section}
	\textbf{Split (\ref{equ:q5 ir}) into a ``dividend yield'' and a ``capital gain'':}
	\[
		1+R^S_{i,t+1}=\underbrace{\frac{\Pi_{i,t+1}+\frac{a}{2}(I_{i,t+1}/A_{i,t+1})^{2}}{1+a\,(I_{it}/A_{it})}}_{\text{dividend yield}\ \approx\ \text{static model}}+\underbrace{\frac{(1-\delta)\big(1+a\,I_{i,t+1}/A_{i,t+1}\big)}{1+a\,(I_{it}/A_{it})}}_{\text{capital gain}\ =\ \text{growth of marginal }q}.
	\]
	\begin{itemize}
		\item \textbf{Dividend yield} --- the squared term is economically tiny, so this piece reproduces the static \textbf{investment} and \textbf{profitability} channels.
		\item \textbf{Capital gain} --- the growth of marginal $q$, roughly proportional to the \textbf{investment-to-assets growth} $\dfrac{I_{i,t+1}/A_{i,t+1}}{I_{it}/A_{it}}$ (Cochrane 1991).
	\end{itemize}
	\begin{block}{The expected return now has \emph{three} ``determinants''}
		It rises with (i) \textbf{low} investment $I_{it}/A_{it}$, (ii) \textbf{high} expected profitability $\mathrm{E}_t[\Pi_{i,t+1}]$, and now (iii) \textbf{high} expected investment growth.
	\end{block}
\end{frame}

\begin{frame}{The Expected-Growth Channel --- The Economics}
	\textbf{Claim.} Holding investment $I_{it}/A_{it}$ and expected profitability fixed, \textbf{expected return increases with expected investment growth}.
	\vspace{3pt}
	\textbf{Why.} The term $1+a\,I_{i,t+1}/A_{i,t+1}$ is next period's marginal cost of investment which, by the first principle, equals next period's \textbf{marginal $q$} --- the present value of all cash flows from one extra unit of assets installed next period. It is itself part of \textit{today's} marginal benefit of investing.
	\begin{block}{The discount-rate logic (exactly parallel to profitability)}
		If expected investment next period is \textbf{high} relative to current investment, the marginal benefit of investing \textit{today} is high. To rationalize the \textbf{low current investment} we are holding fixed, the \textbf{discount rate must be high} to offset it.
	\end{block}
	{\footnotesize \textbf{Profitability channel:} high $\mathrm{E}[\Pi]$ with low $I/A$ $\Rightarrow$ high discount rate. \quad \textbf{Growth channel:} high expected investment growth with low $I/A$ $\Rightarrow$ high discount rate. Same NPV logic, one period further out.}
\end{frame}

\begin{frame}[allowframebreaks]{Estimating Expected Growth}
	Investment $I/A$ and profitability ROE are \textbf{observable}; expected investment growth is \textbf{not}. We must \textbf{forecast} it.
	\begin{itemize}
		\item \textbf{Forecast changes, not the rate.} Firm-level $I/A$ (net asset growth) is frequently negative, so the growth \textit{rate} is ill-defined. Forecast the future \textbf{investment-to-assets change} $d^{\tau}I/A$ ($I/A$ at fiscal year $t{+}\tau$ minus current $I/A$), $\tau=1,2,3$.
		\item \textbf{Method.} Monthly \textbf{Fama--MacBeth} cross-sectional regressions; the expected change $\mathrm{E}_t[d^{\tau}I/A]$ combines the latest predictors with average slopes from a \textbf{120-month rolling window}.
	\end{itemize}
	\textbf{Three predictors, each conceptually motivated:}
	\begin{itemize}
		\item \textbf{Tobin's $q$ ($\log q$):} Hayashi --- marginal $q$ drives investment. Empirically \textbf{weak} on its own ($R^2\approx1\%$).
		\item \textbf{Operating cash flows $C^{op}$} (\href{https://doi.org/10.1016/j.jfineco.2016.03.002}{\textcolor{blue}{Ball et al.\ 2016}}): more internal funds $\Rightarrow$ more investment; and it captures \textbf{intangibles} (R\&D etc.\ are expensed, so earnings understate growth). \textbf{The key predictor.}
		\item \textbf{Change in ROE ($d$ROE):} firms with recent profitability gains raise near-term investment (short-run dynamics).
	\end{itemize}
	\break
	\textbf{Benchmark forecasting regression of $d^{1}I/A$} (WLS, monthly, 1963--2018):
	\begin{center}
		\small
		\begin{tabular}{@{}lcccc@{}}
			\toprule
			& $\log q$ & $C^{op}$ & $d$ROE & $R^2$ \\
			\midrule
			Slope & $-0.03$ & $0.52$ & $0.77$ & $6.4\%$ \\
			$t$   & $(-5.6)$ & $(12.8)$ & $(7.6)$ & \\
			\bottomrule
		\end{tabular}
	\end{center}
	The fitted $\mathrm{E}_t[d^{1}I/A]$ tracks the subsequently \textbf{realized} change closely: at the portfolio level expected and realized changes both run from $\approx-16\%$ (Decile 1) to $\approx+7\%$ (Decile 10), with a cross-sectional correlation of $0.64$.
\end{frame}

\begin{frame}{The Expected Growth Factor \texorpdfstring{$R_{\text{Eg}}$}{REg}}
	\textbf{Construction} (Fama--French protocol): independent $2\times3$ sort on \textbf{size} and \textbf{expected 1-year growth} $\mathrm{E}_t[d^{1}I/A]$ (NYSE 30/40/30 breakpoints, value-weighted); $R_{\text{Eg}}=$ high $-$ low.
	\begin{center}
		\small
		\begin{tabular}{@{}lcc@{}}
			\toprule
			$R_{\text{Eg}}$, monthly 1967--2018 & Estimate & $t$ \\
			\midrule
			Average premium                     & $0.84\%$ & $(10.27)$ \\
			$q$-factor alpha                    & $0.67\%$ & $(9.75)$ \\
			\quad loading on $R_{I/A}$          & $0.21$   & $(4.86)$ \\
			\quad loading on $R_{\text{ROE}}$   & $0.30$   & $(9.13)$ \\
			\bottomrule
		\end{tabular}
	\end{center}
	\begin{itemize}
		\item Premium $t=10.27$ clears the Harvey--Liu--Zhu ($t>3$) multiple-testing hurdle by a mile.
		\item The $q$-factor model \textbf{cannot} span it ($\alpha=0.67\%$, $t=9.75$): a genuinely new return dimension.
		\item \textbf{$C^{op}$ is the engine:} adding a cash-flow factor to $q$ cuts the $R_{\text{Eg}}$ alpha to $0.37\%$; $\log q$ and $d$ROE add little alone.
	\end{itemize}
\end{frame}

\begin{frame}{The \texorpdfstring{$q^5$}{q5} Model}
	\textbf{Augment the four $q$-factors with the expected growth factor:}
	\begin{equation}\label{equ:q5model}
		\resizebox{\linewidth}{!}{$\displaystyle
		\mathrm{E}[R_i]-R_f=\beta^i_{\text{Mkt}}\mathrm{E}[R_{\text{Mkt}}]+\beta^i_{\text{Me}}\mathrm{E}[R_{\text{Me}}]+\beta^i_{I/A}\mathrm{E}[R_{I/A}]+\beta^i_{\text{ROE}}\mathrm{E}[R_{\text{ROE}}]+\beta^i_{\text{Eg}}\mathrm{E}[R_{\text{Eg}}]
		$}
	\end{equation}
	The first four factors are \textbf{identical} to the $q$-factor model; only $R_{\text{Eg}}$ is new.
	\begin{block}{The spanning verdict --- maximum monthly Sharpe ratios (Table V)}
		\begin{center}
			\small
			\begin{tabular}{@{}cccccc@{}}
				\toprule
				$q^5$ & BS6 & FF6c & $q$ & FF6 & FF5 \\
				\midrule
				\textbf{0.63} & 0.48 & 0.43 & 0.42 & 0.37 & 0.32 \\
				\bottomrule
			\end{tabular}
		\end{center}
		$R_{\text{Eg}}$ alone has the \textbf{highest} individual Sharpe ratio of any factor in the study, $0.44$.
	\end{block}
\end{frame}

\begin{frame}{The Horse Race --- 150 Anomalies}
	\textbf{Test assets:} the 150 anomalies significant with NYSE breakpoints \& value weights (HXZ 2019, ``Replicating Anomalies''). $q^5$ is the \textbf{overall best performer}.
	\begin{center}
		\small
		\begin{tabular}{@{}lccccc@{}}
			\toprule
			Across 150 H$-$L deciles & $q^5$ & $q$ & FF6 & FF6c & SY4 \\
			\midrule
			Mean $|\alpha_{H-L}|$ (\%/mo) & \textbf{0.19} & 0.28 & 0.30 & 0.27 & 0.29 \\
			\# $|t|\ge1.96$               & \textbf{23}   & 52   & 74   & 59   & 64 \\
			\# $|t|\ge3$                  & \textbf{6}    & 25   & 37   & 25   & 25 \\
			\# GRS rejections (of 150)    & \textbf{57}   & 101  & 91   & 71   & 87 \\
			\bottomrule
		\end{tabular}
	\end{center}
	\begin{itemize}
		\item Adding \textbf{one} factor more than halves the $q$-model's unexplained anomalies ($52\to23$).
		\item Biggest gains in the \textbf{investment} and \textbf{profitability} categories; momentum is further digested (Roe $+$ Eg).
		\item $q^5$ beats the Fama--French six-factor model (even with $RMW_c$) on \textit{every} metric.
	\end{itemize}
\end{frame}

\begin{frame}{What Expected Growth Fixes}
	\textbf{The anomalies that broke the $q$-model are exactly the growth-driven ones} (H$-$L $\alpha$ in \%/mo, $t$ in parentheses):
	\begin{center}
		\small
		\begin{tabular}{@{}lccc@{}}
			\toprule
			High$-$low decile & $\alpha_q$ & $\alpha_{q^5}$ & $\beta_{\text{Eg}}$ \\
			\midrule
			R\&D-to-market (Rdm)                 & $0.81\ (3.6)$  & $0.27\ (1.2)$  & $+0.84$ \\
			Operating accruals (Oa)              & $-0.57\ (-4.3)$ & $-0.20\ (-1.3)$ & $-0.56$ \\
			Discretionary accruals (Dac)         & $-0.74\ (-5.3)$ & $-0.31\ (-2.2)$ & $-0.64$ \\
			$\Delta$ net financial assets (dFin) & $0.41\ (3.0)$  & $0.14\ (1.0)$  & $+0.40$ \\
			\bottomrule
		\end{tabular}
	\end{center}
	\begin{itemize}
		\item \textbf{R\&D:} expensed today $\Rightarrow$ low current earnings, but builds intangible capital $\Rightarrow$ \textbf{high} future growth $\Rightarrow$ $\beta_{\text{Eg}}>0$.
		\item \textbf{Accruals:} given earnings, high accruals mean \textbf{low cash flows} $\Rightarrow$ low internal funds $\Rightarrow$ \textbf{low} expected growth $\Rightarrow$ $\beta_{\text{Eg}}<0$.
	\end{itemize}
	{\footnotesize One anomaly resists \textit{all} models --- returns around earnings announcements (Abr): $\alpha_{q^5}=0.52\%$ ($t=3.8$).}
\end{frame}

\begin{frame}{Interpretation and a Caveat}
	\begin{block}{Dimension reduction, not (necessarily) a risk story}
		HMXZ read $q^5$ as a \textbf{linear approximation} to the nonlinear firm-level cost of capital in (\ref{equ:q5 ir}) --- a device for \textbf{dimension reduction} (Cochrane 2011), \textit{not} a claim of priced risk. The evidence is \textbf{not inconsistent with mispricing} (e.g.\ the market underrating intangibles, Edmans 2011).
	\end{block}
	\textbf{The honest caveat.} Unlike $I/A$ and ROE, expected growth is \textbf{constructed}: it depends on the predictor list, and \textbf{crucially on operating cash flows $C^{op}$}. Drop $C^{op}$ and the factor weakens. The authors echo Fama--French's worry about factors ``that seem empirically robust but lack theoretical motivation.''
	\vspace{2pt}
	\textit{Still:} the $t=10.27$ premium clears every multiple-testing hurdle, and the cash-flow$\,\to\,$growth link rests on a deep accounting literature (Ball et al.\ 2016; Lev \& Gu 2016).
\end{frame}

\begin{frame}{Closing: What 06 Delivers, and the Handoff to 07}
	\textbf{The spine, in three lines.}
	\begin{enumerate}
		\item Firm optimization $\Rightarrow$ investment return $\Rightarrow$ $R^S=R^I$ (Cochrane identity).
		\item Collapse $\Rightarrow$ $\mathrm{E}_t[1+R^S]=\dfrac{\text{expected profitability}}{1+a\,(I/K)}$ (the Investment CAPM).
		\item Read the cross-section $\Rightarrow$ I/A, ROE, and expected growth $\Rightarrow$ the $q$-factor model and its augmented cousin $q^5$.
	\end{enumerate}
	\vspace{4pt}
	\begin{block}{Handoff to Lecture 07}
		\textit{Which} model wins --- FF5 vs.\ Q4 spanning tests, squared Sharpe ratios, the redundancy of HML, the formal horse race (\href{https://doi.org/10.1093/rof/rfy032}{\textcolor{blue}{HMXZ 2019}}), and the broader Factor Zoo / Factor War --- is the subject of Lecture 07. Here we only built the structural challenger.
	\end{block}
\end{frame}
